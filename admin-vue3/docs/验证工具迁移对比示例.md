# 验证工具迁移对比示例

## 📊 新旧方式对比

### 示例1：邮箱验证

#### ❌ 旧方式 (validate.js)
```javascript
import { validEmail } from '@/utils/validate'

function handleSubmit(email) {
  if (validEmail(email)) {
    console.log('邮箱有效')
    // 问题：只知道 true/false，没有详细错误信息
  } else {
    console.log('邮箱无效')
  }
}
```

#### ✅ 新方式 - 基础函数（兼容）
```typescript
// 无需 import，自动导入！

function handleSubmit(email: string) {
  if (validEmail(email)) {
    console.log('邮箱有效')
  } else {
    console.log('邮箱无效')
  }
}
```

#### 🌟 新方式 - Schema（推荐）
```typescript
function handleSubmit(email: string) {
  const result = safeValidate(emailSchema, email)
  
  if (result.success) {
    console.log('邮箱有效:', result.data)
  } else {
    // 获得详细的错误信息
    console.error('验证失败:', result.error.issues[0].message)
    // 输出: "请输入有效的邮箱地址"
  }
}
```

### 示例2：表单验证

#### ❌ 旧方式
```javascript
import { validEmail, validURL } from '@/utils/validate'

export default {
  data() {
    return {
      form: {
        email: '',
        website: ''
      },
      errors: {}
    }
  },
  methods: {
    validate() {
      this.errors = {}
      
      if (!validEmail(this.form.email)) {
        this.errors.email = '邮箱格式错误' // 需要手动定义错误消息
      }
      
      if (this.form.website && !validURL(this.form.website)) {
        this.errors.website = 'URL格式错误' // 需要手动定义错误消息
      }
      
      return Object.keys(this.errors).length === 0
    }
  }
}
```

#### ✅ 新方式 - Vue 3 + Composition API
```vue
<script setup lang="ts">
import { ref } from 'vue'

const form = ref({
  email: '',
  website: ''
})

const errors = ref<Record<string, string>>({})

function validate() {
  errors.value = {}
  
  // 验证邮箱
  const emailResult = safeValidate(emailSchema, form.value.email)
  if (!emailResult.success) {
    errors.value.email = emailResult.error.issues[0].message
    // 自动获得: "请输入有效的邮箱地址"
  }
  
  // 验证网站（可选）
  if (form.value.website) {
    const urlResult = safeValidate(urlSchema, form.value.website)
    if (!urlResult.success) {
      errors.value.website = urlResult.error.issues[0].message
      // 自动获得: "请输入有效的URL地址"
    }
  }
  
  return Object.keys(errors.value).length === 0
}

function submit() {
  if (validate()) {
    console.log('提交:', form.value)
  }
}
</script>

<template>
  <form @submit.prevent="submit">
    <div>
      <input v-model="form.email" placeholder="邮箱" />
      <span class="error">{{ errors.email }}</span>
    </div>
    
    <div>
      <input v-model="form.website" placeholder="网站" />
      <span class="error">{{ errors.website }}</span>
    </div>
    
    <button type="submit">提交</button>
  </form>
</template>
```

### 示例3：复杂对象验证

#### ❌ 旧方式
```javascript
import { validEmail } from '@/utils/validate'

function validateUser(user) {
  const errors = []
  
  // 需要手动验证每个字段
  if (!user.name || user.name.length < 2) {
    errors.push('姓名至少2个字符')
  }
  
  if (!validEmail(user.email)) {
    errors.push('邮箱格式错误')
  }
  
  if (!user.age || user.age < 18) {
    errors.push('必须年满18岁')
  }
  
  if (!['admin', 'user'].includes(user.role)) {
    errors.push('角色无效')
  }
  
  return {
    valid: errors.length === 0,
    errors
  }
}

// 使用
const result = validateUser(userData)
if (!result.valid) {
  console.error(result.errors) // ['姓名至少2个字符', ...]
}
```

#### ✅ 新方式 - Zod Schema
```typescript
import { z } from 'zod'

// 定义 Schema（可复用）
const UserSchema = z.object({
  name: z.string().min(2, '姓名至少2个字符'),
  email: emailSchema, // 复用已有的 emailSchema
  age: z.number().min(18, '必须年满18岁'),
  role: z.enum(['admin', 'user', 'guest'])
})

// 推断 TypeScript 类型
type User = z.infer<typeof UserSchema>
// 类型: { name: string; email: string; age: number; role: 'admin' | 'user' | 'guest' }

// 验证数据
function validateUser(user: unknown) {
  const result = safeValidate(UserSchema, user)
  
  if (!result.success) {
    // 获得所有验证错误
    result.error.issues.forEach(issue => {
      console.log(`${issue.path.join('.')}: ${issue.message}`)
    })
    // 输出:
    // "name: 姓名至少2个字符"
    // "email: 请输入有效的邮箱地址"
    // ...
  }
  
  return result
}

// 使用
const userData = {
  name: '张',
  email: 'invalid',
  age: 15,
  role: 'superadmin'
}

const result = validateUser(userData)

if (result.success) {
  // TypeScript 知道 result.data 的类型是 User
  console.log('用户:', result.data.name, result.data.email)
}
```

**优势对比：**
- ✅ 代码更简洁（从 20+ 行减少到 5 行）
- ✅ 自动获得 TypeScript 类型
- ✅ 详细的错误信息（包含字段路径）
- ✅ Schema 可复用
- ✅ 易于维护和扩展

### 示例4：API 响应验证

#### ❌ 旧方式
```javascript
import { validEmail } from '@/utils/validate'

async function fetchUsers() {
  const response = await fetch('/api/users')
  const data = await response.json()
  
  // 手动验证响应结构
  if (!data.code || !data.data || !Array.isArray(data.data)) {
    throw new Error('API 响应格式错误')
  }
  
  // 验证每个用户的数据
  const users = data.data.map(user => {
    if (!user.id || !user.name || !validEmail(user.email)) {
      throw new Error('用户数据格式错误')
    }
    return user
  })
  
  return users
}

// 问题：
// 1. 验证逻辑冗长
// 2. 没有类型安全
// 3. 错误信息不详细
```

#### ✅ 新方式
```typescript
import { z } from 'zod'

// 定义用户 Schema
const UserSchema = z.object({
  id: z.number(),
  name: z.string(),
  email: emailSchema,
  age: z.number().optional()
})

// 定义 API 响应 Schema
const UsersResponseSchema = z.object({
  code: z.number(),
  data: z.array(UserSchema),
  message: z.string().optional()
})

// 推断类型
type User = z.infer<typeof UserSchema>
type UsersResponse = z.infer<typeof UsersResponseSchema>

async function fetchUsers(): Promise<User[]> {
  const response = await fetch('/api/users')
  const json = await response.json()
  
  // 一行代码完成验证
  const result = safeValidate(UsersResponseSchema, json)
  
  if (!result.success) {
    // 详细的错误信息
    console.error('API 响应验证失败:', result.error.issues)
    throw new Error('API 响应格式错误')
  }
  
  // result.data 有完整的 TypeScript 类型
  return result.data.data
}

// 使用
const users = await fetchUsers()
// TypeScript 知道 users 的类型是 User[]
users.forEach(user => {
  console.log(user.name, user.email) // 有智能提示
})
```

**优势对比：**
- ✅ 代码量减少 70%
- ✅ 完整的类型安全
- ✅ 自动类型推断
- ✅ 详细的错误定位
- ✅ Schema 可复用于其他地方

### 示例5：自定义验证规则

#### ❌ 旧方式
```javascript
// validate.js 中添加新函数
export function validPhone(phone) {
  return /^1[3-9]\d{9}$/.test(phone)
}

export function validPassword(password) {
  if (password.length < 8) return false
  if (!/[A-Z]/.test(password)) return false
  if (!/[a-z]/.test(password)) return false
  if (!/[0-9]/.test(password)) return false
  return true
}

// 使用
import { validPhone, validPassword } from '@/utils/validate'

if (!validPhone(phone)) {
  ElMessage.error('手机号格式错误') // 手动定义错误消息
}

if (!validPassword(password)) {
  ElMessage.error('密码格式错误') // 不够具体
}
```

#### ✅ 新方式
```typescript
import { z } from 'zod'

// 创建自定义 Schema（可复用）
const phoneSchema = z.string()
  .regex(/^1[3-9]\d{9}$/, '请输入有效的手机号')

const passwordSchema = z.string()
  .min(8, '密码至少8位')
  .regex(/[A-Z]/, '必须包含大写字母')
  .regex(/[a-z]/, '必须包含小写字母')
  .regex(/[0-9]/, '必须包含数字')

// 使用
const phoneResult = safeValidate(phoneSchema, '13800138000')
if (!phoneResult.success) {
  ElMessage.error(phoneResult.error.issues[0].message)
  // 输出: "请输入有效的手机号"
}

const pwdResult = safeValidate(passwordSchema, 'weak')
if (!pwdResult.success) {
  pwdResult.error.issues.forEach(issue => {
    ElMessage.error(issue.message)
  })
  // 输出:
  // "密码至少8位"
  // "必须包含大写字母"
  // "必须包含小写字母"
  // "必须包含数字"
}
```

**优势对比：**
- ✅ 错误消息更具体
- ✅ Schema 可组合和复用
- ✅ 支持链式调用
- ✅ 更容易维护

## 🎯 总结

| 场景 | 旧方式 | 新方式 | 改进 |
|------|--------|--------|------|
| 简单验证 | `validEmail(email)` | `validEmail(email)` | ✅ 自动导入 |
| 表单验证 | 手动验证每个字段 | Schema 一键验证 | ✅ 代码减少 60% |
| 复杂对象 | 大量 if-else | 声明式 Schema | ✅ 代码减少 70% |
| API 验证 | 手动检查结构 | Schema 验证 | ✅ 类型安全 |
| 自定义规则 | 编写新函数 | 链式调用 | ✅ 更易维护 |
| 错误信息 | 手动定义 | 自动生成 | ✅ 更详细 |
| TypeScript | ❌ 无支持 | ✅ 完整支持 | ✅ 类型推断 |

## 💡 迁移建议

1. **新功能** - 直接使用新的 `useValidator`
2. **现有代码** - 逐步迁移，不必一次性全部替换
3. **复杂验证** - 优先使用 Schema 方式
4. **简单验证** - 可以使用基础函数保持简洁
